
select * from prod_data;

-- 1. Manufacture Qty
SELECT SUM(manufactured_qty) AS Total_Manufactured_Qty
FROM prod_data;

-- 2. Rejected Qty
SELECT SUM(rejected_qty) as Total_Rejected_Qty
FROM prod_data;

-- 3. Processed Qty
SELECT SUM(processed_qty) as Total_Processed_Qty
FROM prod_data;

-- 4. Wastage Qty
SELECT SUM(manufactured_qty-processed_qty) as Total_Wastage_Qty
FROM prod_data;

-- 5.Employee wise Rejected Qty
SELECT emp_name,
SUM(rejected_qty) as Total_Rejected_Qty
FROM prod_data
WHERE emp_name != 'NA'
GROUP BY emp_name
ORDER BY Total_Rejected_Qty DESC LIMIT 10;

-- 6. Machine Wise Rejected Qty
SELECT machine_name,
SUM(rejected_qty) as Total_Rejected_Qty
FROM prod_data
WHERE machine_name != '0' AND machine_name IS NOT NULL
GROUP BY machine_name
ORDER BY Total_Rejected_Qty DESC LIMIT 10;

-- 7. Production Comparison Trend
SELECT fiscal_date,
SUM(manufactured_qty) as Total_Manufactured_qty,
SUM(processed_qty) as Total_Processed_qty
FROM prod_data
WHERE fiscal_date is NOT NULL
GROUP BY fiscal_date
ORDER BY fiscal_date;

-- 8. Manufactured vs Rejected
SELECT SUM(manufactured_qty) as Total_Manufactured_qty,
SUM(rejected_qty) as Total_Rejected_qty,
(SUM(rejected_qty)*1.0/SUM(manufactured_qty))*100 AS Rejected_Percentage
FROM prod_data;

-- 9. Department Wise Manufacture vs Rejected
SELECT department_name,
SUM(manufactured_qty) as Total_Manufactured_qty,
SUM(rejected_qty) as Total_Rejected_qty
FROM prod_data
WHERE department_name IS NOT NULL
GROUP BY department_name
ORDER BY Total_Manufactured_qty DESC;

-- 10. Emp Wise Rejected Qty
SELECT emp_name,
SUM(rejected_qty) as Total_Rejected_qty
FROM prod_data
WHERE emp_name != 'NA'
GROUP BY emp_name
ORDER BY Total_Rejected_qty DESC
